<script>
  // Simple State Management for Cash Tender calculations and UX Delight
  const billTotal = 857.00;
  let isSubmitting = false;

  function handleQuickCash(amount) {
    const input = document.getElementById('cash-tendered-input');
    if (input) {
      input.value = amount;
      updateChange(amount);
    }
  }

  function clearTender() {
    const input = document.getElementById('cash-tendered-input');
    if (input) {
      input.value = '';
      updateChange(0);
      input.focus();
    }
  }

  function updateChange(tenderedVal) {
    const tendered = parseFloat(tenderedVal) || 0;
    const diff = tendered - billTotal;
    const changeCard = document.getElementById('change-card');
    const changeVal = document.getElementById('change-amount-val');
    const changeTitle = document.getElementById('change-title');
    const changeHelper = document.getElementById('change-helper-text');
    const statusIcon = document.getElementById('status-chip-icon');

    if (!changeCard || !changeVal) return;

    if (diff >= 0) {
      // Full or Overpaid (Return Change)
      changeCard.className = "bg-secondary-container text-on-secondary-container rounded-xl p-3.5 shadow-sm flex flex-col justify-between transition-colors";
      changeTitle.textContent = "Change to Return";
      changeVal.textContent = "₹" + diff.toFixed(2);
      changeHelper.innerHTML = '<span class="material-symbols-outlined text-[15px]">check_circle</span><span>Give exact change</span>';
      statusIcon.textContent = "currency_exchange";
    } else {
      // Underpaid / Partial / Due Khata
      const remainingDue = Math.abs(diff);
      changeCard.className = "bg-tertiary-fixed text-on-tertiary-fixed rounded-xl p-3.5 shadow-sm flex flex-col justify-between transition-colors";
      changeTitle.textContent = "Balance Due (Khata)";
      changeVal.textContent = "₹" + remainingDue.toFixed(2);
      changeHelper.innerHTML = '<span class="material-symbols-outlined text-[15px]">warning</span><span>Will record to Khata</span>';
      statusIcon.textContent = "pending_actions";
    }
  }

  function setPaymentMode(mode) {
    const modes = ['cash', 'upi', 'card', 'split'];
    modes.forEach(m => {
      const panel = document.getElementById(`panel-${m}`);
      const btn = document.getElementById(`btn-${m}`);
      if (panel) panel.classList.toggle('hidden', m !== mode);
      if (btn) {
        if (m === mode) {

    const unlockBtn = document.getElementById('unlock-cta');
    const resumeBtn = document.getElementById('resume-btn');
    const clearSessionBtn = document.getElementById('clear-session-btn');
    const cashierInput = document.getElementById('cashier-id');

    // PIN visibility toggle
    if (toggleVis && pinInput && eyeIcon) {
      toggleVis.addEventListener('click', () => {
        if (pinInput.type === 'password') {
          pinInput.type = 'text';
          eyeIcon.textContent = 'visibility';
        } else {
          pinInput.type = 'password';
          eyeIcon.textContent = 'visibility_off';
        }
      });
    }

    // Interactive On-screen NumPad
    if (numpad && pinInput) {
      numpad.querySelectorAll('.np-btn').forEach(button => {
        button.addEventListener('click', () => {
          const val = button.textContent.trim();
          if (button.id === 'np-clear') {
            pinInput.value = '';
          } else if (button.id === 'np-backspace' || button.querySelector('.material-symbols-outlined')) {
            pinInput.value = pinInput.value.slice(0, -1);
          } else if (val.length === 1 && !isNaN(val)) {
            if (pinInput.value.length < 6) {
              pinInput.value += val;
            }
          }
        });
      });
    }

    // Toggle Numpad display
    if (toggleNumpadBtn && numpad) {
      toggleNumpadBtn.addEventListener('click', () => {
        numpad.classList.toggle('hidden');
      });
    }

    // Trilingual language pills switcher
    const langBtns = document.querySelectorAll('.lang-btn');
    langBtns.forEach(btn => {
      btn.addEventListener('click', () => {
        langBtns.forEach(b => {
          b.className = 'lang-btn px-2.5 py-1 rounded-full font-label-sm text-label-sm text-on-surface-variant hover:text-on-surface transition-all duration-150';
        });
        btn.className = 'lang-btn px-2.5 py-1 rounded-full font-label-sm text-label-sm bg-primary text-on-primary shadow-sm transition-all duration-150';
      });
    });

    // Fast Shift Resume
    if (resumeBtn && pinInput) {
      resumeBtn.addEventListener('click', () => {
        pinInput.focus();
        resumeBtn.classList.add('scale-95');
        setTimeout(() => resumeBtn.classList.remove('scale-95'), 120);
      });
    }

    // Clear Session & Switch Cashier
    if (clearSessionBtn && cashierInput && pinInput) {
      clearSessionBtn.addEventListener('click', () => {
        cashierInput.value = '';
        pinInput.value = '';
        cashierInput.focus();
      });
    }

    // Hardware F1 & Enter Key Handlers
    window.addEventListener('keydown', (e) => {
      if (e.key === 'F1' || e.key === 'Enter') {
        e.preventDefault();
        unlockBtn.click();
      }
    });

    // Micro tactile bounce for unlock
    if (unlockBtn) {
      unlockBtn.addEventListener('click', () => {
        const originalText = unlockBtn.innerHTML;
        unlockBtn.innerHTML = `

        `;
        setTimeout(() => {
          unlockBtn.innerHTML = originalText;
        }, 1200);
      });
    }

<script>
  function triggerHapticToast(message, iconName = 'check_circle') {
    const toast = document.getElementById('toast');
    const toastMsg = document.getElementById('toastMsg');
    const toastIcon = document.getElementById('toastIcon');
    if (!toast) return;

    toastMsg.textContent = message;
    toastIcon.textContent = iconName;
    toast.classList.remove('opacity-0');
    toast.classList.add('opacity-100');

    setTimeout(() => {
      toast.classList.remove('opacity-100');
      toast.classList.add('opacity-0');
    }, 2400);
  }

  function setFilter(buttonEl, filterType) {
    document.querySelectorAll('.date-chip').forEach(btn => {
      btn.classList.remove('bg-on-surface', 'text-surface');
      btn.classList.add('bg-surface-container', 'text-on-surface-variant');
    });
    buttonEl.classList.remove('bg-surface-container', 'text-on-surface-variant');
    buttonEl.classList.add('bg-on-surface', 'text-surface');

    triggerHapticToast('Loaded ' + buttonEl.innerText.trim() + ' Summary', 'filter_alt');
  }

  function toggleLang() {
    const langCurrent = document.getElementById('langCurrent');
    const langs = ['EN', 'ગુ', 'हि'];
    let idx = langs.indexOf(langCurrent.innerText);
    idx = (idx + 1) % langs.length;
    langCurrent.innerText = langs[idx];
    triggerHapticToast('Language switched to ' + langs[idx], 'translate');
  }

  function printSummarySlip() {
    triggerHapticToast('Printing 80mm Day Slip to BT-01...', 'print');
  }

  function shareWhatsAppReport() {
    triggerHapticToast('Generating & Sending PDF via WhatsApp...', 'send');
  }

  function openReconcileModal() {
    const modal = document.getElementById('reconcileModal');
    if (modal) {
      modal.classList.remove('hidden');
      modal.classList.add('flex');
    }
  }

  function closeReconcileModal() {
    const modal = document.getElementById('reconcileModal');
    if (modal) {
      modal.classList.add('hidden');
      modal.classList.remove('flex');
    }
  }

  function confirmShiftClose() {
    closeReconcileModal();
    triggerHapticToast('Shift #2 Successfully Closed & Verified', 'task_alt');
  }
